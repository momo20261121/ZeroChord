cmake_minimum_required(VERSION 3.16)
file(STRINGS "${CMAKE_CURRENT_SOURCE_DIR}/VERSION.txt" ZEROCHORD_VERSION LIMIT_COUNT 1)
project(ZeroChord VERSION ${ZEROCHORD_VERSION} LANGUAGES C CXX)
set(CMAKE_CXX_STANDARD 17)
set(CMAKE_CXX_STANDARD_REQUIRED ON)
option(ZEROCHORD_DESKTOP "Build the SDL desktop preview" ON)
option(ZEROCHORD_GUI "Build the LVGL application" ON)
set(ZEROCHORD_DEPS "${CMAKE_SOURCE_DIR}/.tools/deps" CACHE PATH "Local dependency sources")
find_package(Threads REQUIRED)

add_library(zerochord_core src/music.cpp src/instrument.cpp src/settings_editor.cpp src/settings_io.cpp src/presets.cpp src/patterns.cpp src/sounds.cpp src/soundfont.cpp)
target_include_directories(zerochord_core PUBLIC src)
target_include_directories(zerochord_core SYSTEM PRIVATE "${ZEROCHORD_DEPS}/tomlplusplus" "${ZEROCHORD_DEPS}/TinySoundFont")
target_compile_options(zerochord_core PRIVATE -Wall -Wextra -Wpedantic)
add_library(zerochord_renderer src/renderer.cpp)
target_link_libraries(zerochord_renderer PUBLIC zerochord_core)
target_include_directories(zerochord_renderer PUBLIC src)
target_include_directories(zerochord_renderer SYSTEM PRIVATE "${ZEROCHORD_DEPS}/TinySoundFont")

enable_testing()
add_executable(music_tests tests/music_tests.cpp)
target_link_libraries(music_tests PRIVATE zerochord_core)
add_test(NAME musical_behavior COMMAND music_tests)
add_executable(settings_tests tests/settings_tests.cpp)
target_link_libraries(settings_tests PRIVATE zerochord_core)
add_test(NAME settings_behavior COMMAND settings_tests)
add_executable(preset_tests tests/preset_tests.cpp)
target_link_libraries(preset_tests PRIVATE zerochord_core)
add_test(NAME preset_behavior COMMAND preset_tests)
add_executable(audio_tests tests/audio_tests.cpp)
target_link_libraries(audio_tests PRIVATE zerochord_renderer)
set(ZEROCHORD_TEST_SOUNDFONT "${CMAKE_SOURCE_DIR}/.tools/soundfonts/TimGM6mb.sf2" CACHE FILEPATH "Sound bank for renderer tests")
add_test(NAME audio_rendering COMMAND audio_tests "${ZEROCHORD_TEST_SOUNDFONT}")
add_executable(sound_tests tests/sound_tests.cpp)
target_link_libraries(sound_tests PRIVATE zerochord_renderer)
add_test(NAME sound_library_and_front_notes COMMAND sound_tests "${ZEROCHORD_TEST_SOUNDFONT}")
add_executable(scenario_tests tests/scenario_tests.cpp)
target_link_libraries(scenario_tests PRIVATE zerochord_core)
add_test(NAME physical_key_scenarios COMMAND scenario_tests)
add_executable(zerochord-render tools/render_demo.cpp)
target_link_libraries(zerochord-render PRIVATE zerochord_renderer)

if(ZEROCHORD_GUI)
    set(LV_BUILD_CONF_PATH "${CMAKE_SOURCE_DIR}/lv_conf.h" CACHE STRING "" FORCE)
    set(CONFIG_LV_BUILD_EXAMPLES OFF CACHE BOOL "" FORCE)
    set(CONFIG_LV_BUILD_DEMOS OFF CACHE BOOL "" FORCE)
    set(CONFIG_LV_USE_THORVG_INTERNAL OFF CACHE BOOL "" FORCE)
    add_subdirectory("${ZEROCHORD_DEPS}/lvgl" "${CMAKE_BINARY_DIR}/lvgl" EXCLUDE_FROM_ALL)
    if(ZEROCHORD_DESKTOP)
        set(SDL_SHARED OFF CACHE BOOL "" FORCE)
        set(SDL_STATIC ON CACHE BOOL "" FORCE)
        set(SDL_TEST OFF CACHE BOOL "" FORCE)
        add_subdirectory("${ZEROCHORD_DEPS}/SDL" "${CMAKE_BINARY_DIR}/sdl" EXCLUDE_FROM_ALL)
    endif()
    add_executable(ZeroChord src/main.cpp src/platform.cpp src/ui.cpp src/audio.cpp src/miniaudio_impl.c)
    set_target_properties(ZeroChord PROPERTIES OUTPUT_NAME M5CardputerZero-ZeroChord)
    target_include_directories(ZeroChord PRIVATE src)
    target_include_directories(ZeroChord SYSTEM PRIVATE "${ZEROCHORD_DEPS}/miniaudio")
    target_compile_definitions(ZeroChord PRIVATE ZEROCHORD_DESKTOP=$<BOOL:${ZEROCHORD_DESKTOP}>)
    target_compile_definitions(ZeroChord PRIVATE ZEROCHORD_VERSION="${PROJECT_VERSION}")
    target_compile_options(ZeroChord PRIVATE -Wall -Wextra)
    target_link_libraries(ZeroChord PRIVATE zerochord_renderer lvgl::lvgl Threads::Threads ${CMAKE_DL_LIBS})
    if(ZEROCHORD_DESKTOP)
        target_link_libraries(ZeroChord PRIVATE SDL2-static)
    else()
        target_link_libraries(ZeroChord PRIVATE m)
        target_link_options(ZeroChord PRIVATE $<$<CONFIG:Release>:-Wl,-s>)
    endif()
endif()
